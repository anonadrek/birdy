> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make Audio-ID via BirdNET-Lite the working premium-only feature in v1.0 — push-to-record 3s, lazy-loaded TFLite model, waveform-thumbnail hero in Match/Disambig/NoBird, Field Journal design language, premium-gated on three layers.

Create `tools/build-birdnet-mapping.py`:
```python
"""Build BirdNET-Lite index→Qid mapping filtered to our 839 European species.

Reads:
  tools/birdnet_labels.txt        (one "Scientific_name_Common name" per line, 6 522 lines)
  shared/content/species_list.yaml (839 species with scientific_name + qid)

Writes:
  composeApp/src/commonMain/composeResources/files/birdnet_lite_to_qid.json

Format:
  {
    "_meta": {
      "generated_for_model_version": "birdnet_lite_v2.4",
      "coverage_pct": 87.5,
      "total_birdnet_classes": 6522,
      "mapped_to_qid": 734
    },
    "mapping": {
      "0": "Q12345",
      "42": "Q67890",
      ...
    }
  }
"""
from __future__ import annotations

import json
import sys
from pathlib import Path

import yaml

ROOT = Path(__file__).resolve().parent.parent
LABELS = ROOT / "tools" / "birdnet_labels.txt"
SPECIES_LIST = ROOT / "shared" / "content" / "species_list.yaml"
OUT = ROOT / "composeApp" / "src" / "commonMain" / "composeResources" / "files" / "birdnet_lite_to_qid.json"


def main() -> int:
    if not LABELS.exists():
        print(f"ERROR: {LABELS} missing — download from BirdNET-Lite v2.4 release", file=sys.stderr)
        return 1

    species_yaml = yaml.safe_load(SPECIES_LIST.read_text(encoding="utf-8"))
    by_sci = {s["scientific_name"]: s["qid"] for s in species_yaml["species"]}

    raw_labels = LABELS.read_text(encoding="utf-8").splitlines()
    mapping: dict[str, str] = {}
    for idx, line in enumerate(raw_labels):
        line = line.strip()
        if not line:
            continue
        # Format: "Scientific_name_Common name" — split on first underscore-pair after sci-name
        # Real BirdNET format uses " " between sci and common; safest split: "_" between binomial words
        # Example: "Parus major_Great Tit" → sci="Parus major", common="Great Tit"
        if "_" not in line:
            continue
        sci, _common = line.split("_", 1)
        sci = sci.strip()
        if sci in by_sci:
            mapping[str(idx)] = by_sci[sci]

    coverage = (len(mapping) / len(by_sci)) * 100
    out_payload = {
        "_meta": {
            "generated_for_model_version": "birdnet_lite_v2.4",
            "coverage_pct": round(coverage, 1),
            "total_birdnet_classes": len(raw_labels),
            "mapped_to_qid": len(mapping),
            "total_species_in_list": len(by_sci),
        },
        "mapping": mapping,
    }

    OUT.parent.mkdir(parents=True, exist_ok=True)
    OUT.write_text(json.dumps(out_payload, ensure_ascii=False, indent=2), encoding="utf-8")
    print(f"Wrote {OUT}: {len(mapping)} mappings, coverage {coverage:.1f}%")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```kotlin
package se.birdy.ml.tools

import com.charleskorn.kaml.Yaml
import kotlinx.serialization.Serializable
import kotlinx.serialization.json.Json
import kotlinx.serialization.json.JsonObject
import kotlinx.serialization.json.jsonObject
import kotlinx.serialization.json.jsonPrimitive
import java.io.File
import kotlin.system.exitProcess

@Serializable
private data class SpeciesEntry(val qid: String, val scientific_name: String)

@Serializable
private data class SpeciesList(val species: List<SpeciesEntry>)

fun main(args: Array<String>) {
    require(args.size == 2) { "Usage: ValidateBirdNetMapping <mapping.json> <species_list.yaml>" }
    val mappingFile = File(args[0])
    val speciesFile = File(args[1])

    val errors = mutableListOf<String>()

    val mappingJson = Json.parseToJsonElement(mappingFile.readText()).jsonObject
    val meta = mappingJson["_meta"]?.jsonObject ?: run {
        errors += "_meta block missing in ${mappingFile.name}"
        return abort(errors)
    }
    val mapping = mappingJson["mapping"]?.jsonObject ?: run {
        errors += "mapping block missing"
        return abort(errors)
    }

    val coverage = meta["coverage_pct"]?.jsonPrimitive?.content?.toDoubleOrNull() ?: 0.0
    if (coverage < 70.0) errors += "coverage_pct $coverage < 70.0 threshold"

    val speciesYaml = Yaml.default.decodeFromString(SpeciesList.serializer(), speciesFile.readText())
    val validQids = speciesYaml.species.map { it.qid }.toSet()

    val mapped = mapping.values.map { it.jsonPrimitive.content }
    val invalidQids = mapped.filterNot { it in validQids }
    if (invalidQids.isNotEmpty()) {
        errors += "Mapping references ${invalidQids.size} qids not in species_list: ${invalidQids.take(5)}"
    }

    val duplicateIndices = mapping.keys.groupingBy { it }.eachCount().filter { it.value > 1 }
    if (duplicateIndices.isNotEmpty()) errors += "Duplicate indices: $duplicateIndices"

    if (errors.isNotEmpty()) abort(errors)
    println("✓ BirdNET mapping valid: ${mapped.size} entries, coverage $coverage%")
}

private fun abort(errors: List<String>) {
    System.err.println("BirdNET mapping validation FAILED:")
    errors.forEach { System.err.println("  ✗ $it") }
    exitProcess(1)
}
```

```kotlin
package se.birdy.ml

data class AudioInput(
    val waveform: FloatArray,
    val sampleRate: Int,
    val durationMs: Int,
    val rawPcm: ShortArray? = null,
) {
    init {
        require(waveform.isNotEmpty()) { "waveform must not be empty" }
        require(sampleRate > 0) { "sampleRate must be positive" }
    }

    override fun equals(other: Any?): Boolean {
        if (this === other) return true
        if (other !is AudioInput) return false
        return waveform.contentEquals(other.waveform) &&
            sampleRate == other.sampleRate &&
            durationMs == other.durationMs &&
            (rawPcm?.contentEquals(other.rawPcm) ?: (other.rawPcm == null))
    }

    override fun hashCode(): Int {
        var result = waveform.contentHashCode()
        result = 31 * result + sampleRate
        result = 31 * result + durationMs
        result = 31 * result + (rawPcm?.contentHashCode() ?: 0)
        return result
    }
}
```

```kotlin
package se.birdy.ml

import kotlinx.coroutines.sync.Mutex
import kotlinx.coroutines.sync.withLock

class SessionFailureGuard<T : AutoCloseable>(
    private val primary: T,
    private val fallback: T,
    private val onDegrade: () -> Unit,
    private val strikeThreshold: Int = 3,
) : AutoCloseable {
    private val mutex = Mutex()
    private var strikes = 0
    private var degraded = false
    private var closed = false

    val current: T
        get() = if (degraded) fallback else primary

    suspend fun <R> withSession(block: suspend (T) -> R): R = mutex.withLock {
        check(!closed) { "SessionFailureGuard closed" }
        val active = current
        try {
            val r = block(active)
            strikes = 0
            r
        } catch (t: Throwable) {
            if (!degraded) {
                strikes++
                if (strikes >= strikeThreshold) {
                    degraded = true
                    runCatching { primary.close() }
                    onDegrade()
                }
            }
            throw t
        }
    }

    override fun close() {
        if (closed) return
        closed = true
        runCatching { primary.close() }
        runCatching { fallback.close() }
    }
}
```

```kotlin
package se.birdy.ml

import android.content.Context
import kotlinx.coroutines.sync.Mutex
import kotlinx.coroutines.sync.withLock
import org.tensorflow.lite.Interpreter
import java.io.FileInputStream
import java.nio.ByteBuffer
import java.nio.ByteOrder
import java.nio.MappedByteBuffer
import java.nio.channels.FileChannel

class AndroidTfliteAudioRunner private constructor(
    private val interpreter: Interpreter,
    private val mapper: BirdNetLabelMapper,
    override val info: AudioModelInfo,
) : BirdAudioClassifier {
    private val mutex = Mutex()
    private var closed = false

    override suspend fun classify(input: AudioInput): AudioClassification = mutex.withLock {
        check(!closed) { "AndroidTfliteAudioRunner closed" }
        require(input.waveform.size == 144_000) {
            "Expected 144000 samples, got ${input.waveform.size}"
        }

        val inputBuf = ByteBuffer.allocateDirect(144_000 * 4).order(ByteOrder.nativeOrder())
        input.waveform.forEach { inputBuf.putFloat(it) }
        inputBuf.rewind()

        val outputClasses = info.outputShape.last()
        val outputBuf = ByteBuffer.allocateDirect(outputClasses * 4).order(ByteOrder.nativeOrder())

        val t0 = System.currentTimeMillis()
        interpreter.run(inputBuf, outputBuf)
        val inferenceMs = System.currentTimeMillis() - t0

        outputBuf.rewind()
        val scores = FloatArray(outputClasses) { outputBuf.float }

        val top3 = scores.mapIndexed { idx, score -> idx to score }
            .sortedByDescending { it.second }
            .take(3)
            .mapNotNull { (idx, score) ->
                mapper.lookup(idx)?.let { qid -> Classification(qid, score) }
            }

        AudioClassification(top = top3, inferenceMs = inferenceMs, modelVersion = info.modelVersion)
    }

    override fun close() {
        if (closed) return
        closed = true
        runCatching { interpreter.close() }
    }

    companion object {
        suspend fun load(context: Context, modelAssetPath: String = "models/birdnet_lite_v2.tflite"): AndroidTfliteAudioRunner {
            val model = loadModelFile(context, modelAssetPath)
            val options = Interpreter.Options().apply {
                numThreads = 4
            }
            val interpreter = Interpreter(model, options)

            val inputShape = interpreter.getInputTensor(0).shape().toList()
            val outputShape = interpreter.getOutputTensor(0).shape().toList()
            val mapper = BirdNetLabelMapper.load()
            val info = AudioModelInfo(
                modelVersion = mapper.modelVersion,
                inputShape = inputShape,
                outputShape = outputShape,
                coveragePct = mapper.coveragePct,
            )
            return AndroidTfliteAudioRunner(interpreter, mapper, info)
        }

        private fun loadModelFile(context: Context, assetPath: String): MappedByteBuffer {
            val fd = context.assets.openFd(assetPath)
            FileInputStream(fd.fileDescriptor).use { stream ->
                val channel = stream.channel
                return channel.map(FileChannel.MapMode.READ_ONLY, fd.startOffset, fd.declaredLength)
            }
        }
    }
}
```

```kotlin
package se.birdy.ml

import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow

sealed interface AudioClassifierBootstrapState {
    object Idle : AudioClassifierBootstrapState
    object Loading : AudioClassifierBootstrapState
    data class Ready(val classifier: BirdAudioClassifier, val mode: AudioClassifierMode) : AudioClassifierBootstrapState
    data class Failed(val cause: Throwable) : AudioClassifierBootstrapState
}

class AudioClassifierBootstrap(
    private val build: suspend () -> Pair<BirdAudioClassifier, AudioClassifierMode>,
) {
    private val _state = MutableStateFlow<AudioClassifierBootstrapState>(AudioClassifierBootstrapState.Idle)
    val state: StateFlow<AudioClassifierBootstrapState> = _state

    suspend fun ensureReady(): BirdAudioClassifier {
        val current = _state.value
        if (current is AudioClassifierBootstrapState.Ready) return current.classifier
        _state.value = AudioClassifierBootstrapState.Loading
        return try {
            val (clf, mode) = build()
            _state.value = AudioClassifierBootstrapState.Ready(clf, mode)
            clf
        } catch (t: Throwable) {
            _state.value = AudioClassifierBootstrapState.Failed(t)
            throw t
        }
    }

    fun release() {
        (_state.value as? AudioClassifierBootstrapState.Ready)?.classifier?.close()
        _state.value = AudioClassifierBootstrapState.Idle
    }
}
```

```kotlin
package se.birdy.ml

import android.Manifest
import android.annotation.SuppressLint
import android.media.AudioFormat
import android.media.AudioRecord
import android.media.MediaRecorder
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import kotlin.math.sqrt

class AndroidAudioRecorder(
    val sampleRate: Int = 48_000,
    val durationMs: Int = 3_000,
) {
    val expectedSamples: Int = sampleRate * durationMs / 1000

    @SuppressLint("MissingPermission")
    suspend fun record3s(onLevel: (rms: Float) -> Unit): ShortArray = withContext(Dispatchers.IO) {
        val minBuf = AudioRecord.getMinBufferSize(
            sampleRate,
            AudioFormat.CHANNEL_IN_MONO,
            AudioFormat.ENCODING_PCM_16BIT,
        )
        require(minBuf > 0) { "AudioRecord.getMinBufferSize returned $minBuf" }

        val source = pickAudioSource()
        val recorder = AudioRecord(
            source,
            sampleRate,
            AudioFormat.CHANNEL_IN_MONO,
            AudioFormat.ENCODING_PCM_16BIT,
            maxOf(minBuf, expectedSamples * 2),
        )

        if (recorder.state != AudioRecord.STATE_INITIALIZED) {
            recorder.release()
            error("AudioRecord failed to initialize (state=${recorder.state})")
        }

        try {
            recorder.startRecording()
            val buffer = ShortArray(expectedSamples)
            var totalRead = 0
            val chunkSize = sampleRate / 30 // ~30 Hz RMS callback

            while (totalRead < expectedSamples) {
                val toRead = minOf(chunkSize, expectedSamples - totalRead)
                val read = recorder.read(buffer, totalRead, toRead)
                if (read <= 0) error("AudioRecord.read returned $read")

                val rms = computeRms(buffer, totalRead, read)
                onLevel(rms)
                totalRead += read
            }
            buffer
        } finally {
            runCatching { recorder.stop() }
            recorder.release()
        }
    }

    private fun pickAudioSource(): Int {
        // UNPROCESSED gives cleanest signal for ML on supported devices; fallback to VOICE_RECOGNITION
        return try {
            MediaRecorder.AudioSource.UNPROCESSED
        } catch (_: Throwable) {
            MediaRecorder.AudioSource.VOICE_RECOGNITION
        }
    }

    private fun computeRms(buffer: ShortArray, offset: Int, length: Int): Float {
        if (length == 0) return 0f
        var sum = 0.0
        for (i in offset until offset + length) {
            val s = buffer[i] / 32768.0
            sum += s * s
        }
        return sqrt(sum / length).toFloat().coerceIn(0f, 1f)
    }
}
```

```kotlin
    private val activity: ComponentActivity,
) : AudioPermissionController, DefaultLifecycleObserver {
    private val _state = MutableStateFlow(PermissionState.Unknown)
    override val state: StateFlow<PermissionState> = _state

    private val launcher: ActivityResultLauncher<String> = activity.registerForActivityResult(
        ActivityResultContracts.RequestPermission()
    ) { granted ->
        _state.value = if (granted) PermissionState.Granted else {
            if (ActivityCompat.shouldShowRequestPermissionRationale(activity, Manifest.permission.RECORD_AUDIO)) {
                PermissionState.Denied
            } else {
                PermissionState.PermanentlyDenied
            }
        }
    }

        recheck()
    }

        recheck()
    }

        launcher.launch(Manifest.permission.RECORD_AUDIO)
    }

    override fun openSettings() {
        val intent = Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS).apply {
            data = Uri.fromParts("package", activity.packageName, null)
        }
        activity.startActivity(intent)
    }

    override fun recheck() {
        val granted = ContextCompat.checkSelfPermission(activity, Manifest.permission.RECORD_AUDIO) ==
            PackageManager.PERMISSION_GRANTED
        _state.value = when {
            granted -> PermissionState.Granted
            ActivityCompat.shouldShowRequestPermissionRationale(activity, Manifest.permission.RECORD_AUDIO) ->
                PermissionState.Denied
            _state.value == PermissionState.Unknown -> PermissionState.Unknown
            else -> PermissionState.PermanentlyDenied
        }
```

```kotlin
package se.birdy.app.ui.audio

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.Job
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
import kotlinx.serialization.json.Json
import se.birdy.ml.AudioInput
import se.birdy.ml.BirdAudioClassifier
import se.birdy.ml.Classification
import se.birdy.ml.ScanSource
import se.birdy.ml.normalize

class AudioScanViewModel(
    private val classifierProvider: suspend () -> BirdAudioClassifier,
    private val recorder: AudioRecorderApi,
    private val waveformRenderer: WaveformRendererApi,
    private val audioStorageDir: () -> String,
    private val clock: () -> Long = { System.currentTimeMillis() },
) : ViewModel() {
    private val _state = MutableStateFlow<AudioScanState>(AudioScanState.Preparing)
    val state: StateFlow<AudioScanState> = _state

    private var recordingJob: Job? = null

    fun onPermissionState(p: PermissionState) {
        when (p) {
            PermissionState.Granted -> if (_state.value !is AudioScanState.Recording) {
                _state.value = AudioScanState.Idle
            }
            PermissionState.Denied -> _state.value = AudioScanState.PermissionNeeded(canRequest = true)
            PermissionState.PermanentlyDenied -> _state.value = AudioScanState.Error.PermanentlyDenied
            PermissionState.Unknown -> _state.value = AudioScanState.Preparing
        }
    }

    fun startRecording() {
        val current = _state.value
        if (current !is AudioScanState.Idle) return
        _state.value = AudioScanState.Recording(rms = 0f, elapsedMs = 0L)
        val t0 = clock()
        recordingJob = viewModelScope.launch {
            try {
                val pcm = recorder.record3s(
                    onLevel = { rms ->
                        val elapsed = clock() - t0
                        val s = _state.value
                        if (s is AudioScanState.Recording) {
                            _state.value = AudioScanState.Recording(rms, elapsed)
                        }
                    }
                )
                val frozen = (_state.value as? AudioScanState.Recording)?.rms ?: 0f
                _state.value = AudioScanState.Analyzing(rmsFrozen = frozen)
                analyzeAndNavigate(pcm)
            } catch (t: Throwable) {
                _state.value = AudioScanState.Error.RecordingFailed
            }
        }
    }

    fun cancelRecording() {
        recordingJob?.cancel()
        recordingJob = null
        if (_state.value !is AudioScanState.Error) _state.value = AudioScanState.Idle
    }

    private suspend fun analyzeAndNavigate(pcm: ShortArray) {
        val classifier = runCatching { classifierProvider() }.getOrElse {
            _state.value = AudioScanState.Error.BootstrapFailed(it.message ?: "bootstrap failed")
            return
        }
        val ts = clock()
        val pngPath = withContext(Dispatchers.IO) {
            waveformRenderer.renderWaveformPng(pcm, "${audioStorageDir()}/$ts.png")
        }
        val audioPath = withContext(Dispatchers.IO) {
            waveformRenderer.encodeOpus(pcm, "${audioStorageDir()}/$ts.opus")
        }
        val waveform = normalize(pcm)
        val classification = classifier.classify(AudioInput(waveform, 48_000, 3_000, rawPcm = pcm))

        if (classification.top.isEmpty()) {
            val source = ScanSource.Audio(
                frameJpegPath = pngPath,
                classification = Classification(qid = "", confidence = 0f),
                audioWavPath = audioPath,
            )
            _state.value = AudioScanState.NavigateToMatch(Json.encodeToString(ScanSourceSerialization.serializer(), source.toSerial()))
            return
        }

        val top = classification.top.first()
        val source = ScanSource.Audio(
            frameJpegPath = pngPath,
            classification = top,
            audioWavPath = audioPath,
        )
        _state.value = AudioScanState.NavigateToMatch(Json.encodeToString(ScanSourceSerialization.serializer(), source.toSerial()))
    }
}

interface AudioRecorderApi {
    suspend fun record3s(onLevel: (Float) -> Unit): ShortArray
}

interface WaveformRendererApi {
    suspend fun renderWaveformPng(pcm: ShortArray, outPath: String): String
    suspend fun encodeOpus(pcm: ShortArray, outPath: String): String
}
```

```kotlin
package se.birdy.app.ui.audio

import android.graphics.*
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.File
import java.io.FileOutputStream
import kotlin.math.abs

actual class WaveformRenderer actual constructor() {
    actual suspend fun renderWaveformPng(pcm: ShortArray, outPath: String): String = withContext(Dispatchers.Default) {
        val width = 600
        val height = 200
        val buckets = 120
        val samplesPerBucket = pcm.size / buckets

        val peaks = FloatArray(buckets) { b ->
            var max = 0
            val start = b * samplesPerBucket
            val end = (start + samplesPerBucket).coerceAtMost(pcm.size)
            for (i in start until end) {
                val v = abs(pcm[i].toInt())
                if (v > max) max = v
            }
            max / 32768f
        }
        // 3-tap smoothing
        val smoothed = FloatArray(buckets) { i ->
            val prev = peaks[(i - 1).coerceAtLeast(0)]
            val cur = peaks[i]
            val next = peaks[(i + 1).coerceAtMost(buckets - 1)]
            (prev + cur + next) / 3f
        }

        val bitmap = Bitmap.createBitmap(width, height, Bitmap.Config.ARGB_8888)
        val canvas = Canvas(bitmap)
        canvas.drawColor(Color.parseColor("#EFE7D6")) // PaperBg

        val barPaint = Paint().apply {
            color = Color.parseColor("#3F4F30") // MarginaliaInk
            style = Paint.Style.FILL
            isAntiAlias = true
        }
        val underlinePaint = Paint().apply {
            color = Color.parseColor("#A8552D") // AccentCopper
            style = Paint.Style.STROKE
            strokeWidth = 2f
            isAntiAlias = true
        }

        val barWidth = width.toFloat() / buckets
        val centerY = height / 2f
        val maxHalfHeight = height * 0.4f
        smoothed.forEachIndexed { i, level ->
            val h = (maxHalfHeight * level).coerceAtLeast(2f)
            val x = i * barWidth
            canvas.drawRect(x + 1f, centerY - h, x + barWidth - 1f, centerY + h, barPaint)
        }
        canvas.drawLine(0f, height - 6f, width.toFloat(), height - 6f, underlinePaint)

        val file = File(outPath)
        file.parentFile?.mkdirs()
        FileOutputStream(file).use { out ->
            bitmap.compress(Bitmap.CompressFormat.PNG, 100, out)
        }
        bitmap.recycle()
        outPath
    }
}
```

```kotlin
package se.birdy.app.ui.audio

import android.media.MediaCodec
import android.media.MediaCodecInfo
import android.media.MediaFormat
import android.media.MediaMuxer
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.io.File
import java.nio.ByteBuffer

actual class OpusEncoder actual constructor() {
    actual suspend fun encodeOpus(pcm: ShortArray, outPath: String, sampleRate: Int): String =
        withContext(Dispatchers.Default) {
            val file = File(outPath)
            file.parentFile?.mkdirs()

            val format = MediaFormat.createAudioFormat(MediaFormat.MIMETYPE_AUDIO_OPUS, sampleRate, 1).apply {
                setInteger(MediaFormat.KEY_BIT_RATE, 32_000)
                setInteger(MediaFormat.KEY_AAC_PROFILE, MediaCodecInfo.CodecProfileLevel.AACObjectLC)
            }
            val codec = MediaCodec.createEncoderByType(MediaFormat.MIMETYPE_AUDIO_OPUS).apply {
                configure(format, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE)
                start()
            }
            val muxer = MediaMuxer(outPath, MediaMuxer.OutputFormat.MUXER_OUTPUT_OGG)

            val pcmBytes = ByteBuffer.allocate(pcm.size * 2).apply {
                order(java.nio.ByteOrder.LITTLE_ENDIAN)
                pcm.forEach { putShort(it) }
            }.array()

            var trackIdx = -1
            var muxerStarted = false
            var inputOffset = 0
            var sawEos = false
            val bufferInfo = MediaCodec.BufferInfo()

            while (!sawEos) {
                val inIdx = codec.dequeueInputBuffer(10_000)
                if (inIdx >= 0) {
                    val inputBuf = codec.getInputBuffer(inIdx)!!
                    inputBuf.clear()
                    val remaining = pcmBytes.size - inputOffset
                    if (remaining <= 0) {
                        codec.queueInputBuffer(inIdx, 0, 0, 0, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
                    } else {
                        val chunk = remaining.coerceAtMost(inputBuf.capacity())
                        inputBuf.put(pcmBytes, inputOffset, chunk)
                        val presUs = (inputOffset.toLong() * 1_000_000L / 2L / sampleRate)
                        codec.queueInputBuffer(inIdx, 0, chunk, presUs, 0)
                        inputOffset += chunk
                    }
                }
                val outIdx = codec.dequeueOutputBuffer(bufferInfo, 10_000)
                when {
                    outIdx == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED -> {
                        trackIdx = muxer.addTrack(codec.outputFormat)
                        muxer.start()
                        muxerStarted = true
                    }
                    outIdx >= 0 -> {
                        val out = codec.getOutputBuffer(outIdx)!!
                        if (muxerStarted && bufferInfo.size > 0) {
                            muxer.writeSampleData(trackIdx, out, bufferInfo)
                        }
                        if ((bufferInfo.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) {
                            sawEos = true
                        }
                        codec.releaseOutputBuffer(outIdx, false)
                    }
                }
            }
            runCatching { codec.stop() }
            codec.release()
            if (muxerStarted) {
                runCatching { muxer.stop() }
            }
            muxer.release()
            outPath
        }
}
```

```kotlin
class ListenLauncherViewModel(
    private val premiumState: StateFlow<PremiumState>,

                is PremiumState.Loading -> { /* pessimistic lock — no-op */ }

    }
```

```python
"""Build 30-sample audio corpus from xeno-canto CC-licensed recordings.

Usage: python tools/ml-eval/build_audio_corpus.py

Requires: ffmpeg on PATH (for mp3 → 48 kHz mono PCM WAV resample).
"""
from __future__ import annotations

import json
import shutil
import subprocess
import sys
import time
import urllib.parse
import urllib.request
import wave
from pathlib import Path

import numpy as np
import yaml

ROOT = Path(__file__).resolve().parent.parent.parent
SPECIES_LIST = ROOT / "shared" / "content" / "species_list.yaml"
OUT_DIR = ROOT / "tools" / "ml-eval" / "corpus_audio"

TARGET_SCI_NAMES = [
    "Parus major", "Cyanistes caeruleus", "Turdus merula", "Turdus philomelos",
    "Erithacus rubecula", "Fringilla coelebs", "Carduelis carduelis", "Sturnus vulgaris",
    "Corvus corone", "Pica pica", "Sitta europaea", "Periparus ater",
    "Phylloscopus collybita", "Phylloscopus trochilus", "Sylvia atricapilla", "Troglodytes troglodytes",
    "Regulus regulus", "Aegithalos caudatus", "Pyrrhula pyrrhula", "Loxia curvirostra",
    "Buteo buteo", "Accipiter nisus", "Dendrocopos major", "Picus viridis",
    "Apus apus", "Hirundo rustica", "Delichon urbicum", "Motacilla alba",
    "Anthus pratensis", "Anas platyrhynchos",
]


def search_xc(sci_name: str) -> dict | None:
    query = urllib.parse.quote(f'{sci_name} type:song q:A len:5-30 cnt:Sweden')
    url = f"https://xeno-canto.org/api/2/recordings?query={query}"
    with urllib.request.urlopen(url, timeout=20) as resp:
        data = json.loads(resp.read())
    for rec in data.get("recordings", []):
        lic = rec.get("lic", "")
        if "by" in lic or "publicdomain" in lic or "zero" in lic:
            return rec
    return None


def ffmpeg_to_wav(src: Path, dst: Path) -> bool:
    ffmpeg = shutil.which("ffmpeg")
    if not ffmpeg:
        raise RuntimeError("ffmpeg not on PATH")
    result = subprocess.run(
        [ffmpeg, "-y", "-i", str(src), "-ac", "1", "-ar", "48000", "-f", "wav", str(dst)],
        capture_output=True,
        check=False,
    )
    return result.returncode == 0


def download_and_crop(rec: dict, qid: str, sci: str) -> bool:
    url = rec.get("file") or rec.get("file-name")
    if not url:
        return False
    if url.startswith("//"):
        url = "https:" + url
    raw_path = OUT_DIR / f"_raw_{qid}.mp3"
    tmp_wav = OUT_DIR / f"_tmp_{qid}.wav"
    out_path = OUT_DIR / f"{qid}_{sci.replace(' ', '_')}.wav"

    urllib.request.urlretrieve(url, raw_path)
    if not ffmpeg_to_wav(raw_path, tmp_wav):
        return False

    with wave.open(str(tmp_wav), "rb") as w:
        frames = w.readframes(w.getnframes())
        rate = w.getframerate()
    pcm = np.frombuffer(frames, dtype=np.int16)
    window = 3 * rate
    if len(pcm) < window:
        pcm = np.pad(pcm, (0, window - len(pcm)))

    step = rate
    best_rms = -1.0
    best_start = 0
    for start in range(0, max(1, len(pcm) - window), step):
        chunk = pcm[start:start + window].astype(np.float32) / 32768.0
        rms = float(np.sqrt(np.mean(chunk * chunk)))
        if rms > best_rms:
            best_rms = rms
            best_start = start
    clipped = pcm[best_start:best_start + window]

    with wave.open(str(out_path), "wb") as w:
        w.setnchannels(1)
        w.setsampwidth(2)
        w.setframerate(48_000)
        w.writeframes(clipped.tobytes())

    raw_path.unlink(missing_ok=True)
    tmp_wav.unlink(missing_ok=True)
    return True


def main() -> int:
    OUT_DIR.mkdir(parents=True, exist_ok=True)
    species = yaml.safe_load(SPECIES_LIST.read_text(encoding="utf-8"))["species"]
    by_sci = {s["scientific_name"]: s["qid"] for s in species}
    saved = 0
    for sci in TARGET_SCI_NAMES:
        qid = by_sci.get(sci)
        if not qid:
            print(f"skip — {sci} not in species_list", file=sys.stderr)
            continue
        try:
            rec = search_xc(sci)
            if rec and download_and_crop(rec, qid, sci):
                saved += 1
                print(f"✓ {sci} ({qid})")
            else:
                print(f"✗ {sci} — no CC recording found")
        except Exception as e:
            print(f"✗ {sci} — {e}", file=sys.stderr)
        time.sleep(1.0)
    print(f"\nCorpus built: {saved} samples in {OUT_DIR}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Run BirdNET-Lite on the corpus, compute top-1 and top-3 accuracy."""
from __future__ import annotations

import json
import sys
import wave
from pathlib import Path

import numpy as np
import tflite_runtime.interpreter as tflite

ROOT = Path(__file__).resolve().parent.parent.parent
MODEL = ROOT / "composeApp" / "src" / "androidMain" / "assets" / "models" / "birdnet_lite_v2.tflite"
MAPPING = ROOT / "composeApp" / "src" / "commonMain" / "composeResources" / "files" / "birdnet_lite_to_qid.json"
CORPUS = ROOT / "tools" / "ml-eval" / "corpus_audio"


def load_wav(path: Path) -> np.ndarray:
    with wave.open(str(path), "rb") as w:
        frames = w.readframes(w.getnframes())
    return np.frombuffer(frames, dtype=np.int16).astype(np.float32) / 32768.0


def main() -> int:
    interp = tflite.Interpreter(model_path=str(MODEL), num_threads=4)
    interp.allocate_tensors()
    in_det = interp.get_input_details()[0]
    out_det = interp.get_output_details()[0]

    mapping_payload = json.loads(MAPPING.read_text(encoding="utf-8"))
    mapping = {int(k): v for k, v in mapping_payload["mapping"].items()}

    samples = sorted(CORPUS.glob("Q*.wav"))
    if not samples:
        print("No samples found in corpus_audio/", file=sys.stderr)
        return 1

    top1 = 0
    top3 = 0
    for path in samples:
        true_qid = path.stem.split("_")[0]
        waveform = load_wav(path)
        if len(waveform) < 144_000:
            waveform = np.pad(waveform, (0, 144_000 - len(waveform)))
        waveform = waveform[:144_000].astype(np.float32)
        interp.set_tensor(in_det["index"], waveform.reshape(in_det["shape"]))
        interp.invoke()
        scores = interp.get_tensor(out_det["index"]).flatten()

        ranked = np.argsort(-scores)
        top3_qids = [mapping.get(int(i)) for i in ranked[:50] if mapping.get(int(i)) is not None][:3]
        if top3_qids and top3_qids[0] == true_qid:
            top1 += 1
        if true_qid in top3_qids:
            top3 += 1

    n = len(samples)
    print(f"\nTop-1: {top1}/{n} = {100*top1/n:.1f}%")
    print(f"Top-3: {top3}/{n} = {100*top3/n:.1f}%")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```markdown
- **Lazy bootstrap pattern (AtomicReference + CAS):** `private val audioBootstrapCache = AtomicReference<Deferred<BirdAudioClassifier>?>(null)`; `audioBootstrapCache.get() ?: GlobalScope.async(start = LAZY) { ... }.also { audioBootstrapCache.compareAndSet(null, it).let { ok -> if (!ok) it.cancel() } }.await()`.
```
